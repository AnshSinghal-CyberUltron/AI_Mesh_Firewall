"""Run openai-conformance's LGW01 step VERBATIM (from gateway-v2.yml at a git ref) against
scratch conformance suites with real pytest + pytest-timeout, and report GREEN/RED.

Reviewer B's conf-hang reproduction (PR #137 S2), extended. Usage:

Only change to the step text: /tmp/gw01- -> <scratch>/gw01- (no clash with other agents).
The scratch tree mirrors the job: <root>/.venv/bin/python (venv-ci: pytest 9.2.0,

def git_show(repo: str, ref: str, path: str) -> str | None:
    proc = subprocess.run(
        ["git", "-C", repo, "show", f"{ref}:{path}"], capture_output=True, text=True
    )
    return proc.stdout if proc.returncode == 0 else None


def step_text(repo: str, ref: str) -> str:
    wf = yaml.safe_load(git_show(repo, ref, ".github/workflows/gateway-v2.yml") or "")
    for step in wf["jobs"]["openai-conformance"]["steps"]:
        if str(step.get("name", "")).startswith("LGW01-1/2/4/5"):
            return str(step["run"])
    raise SystemExit("LGW01 step not found")


def build(root: Path, repo: str, ref: str, scenario: str) -> None:
    extra, conftest, _ = SCENARIOS[scenario]
    (root / ".venv/bin").mkdir(parents=True)
    py = root / ".venv/bin/python"
    py.write_text(f'#!/bin/sh\nexec {VENV_PY} "$@"\n')
    py.chmod(0o755)
    gw = root / "gateway_v2"
    (gw / "lint").mkdir(parents=True)
    (gw / "lint/__init__.py").write_text("")
    (gw / "lint/compare_junit.py").write_text(
        git_show(repo, ref, "gateway_v2/lint/compare_junit.py") or ""
    )
    (gw / "pyproject.toml").write_text(PYPROJECT)
    tests = gw / "tests/openai_conformance"
    tests.mkdir(parents=True)
    (tests / "test_conf.py").write_text(BASE + "\n\n" + extra)
    if conftest:
        (tests / "conftest.py").write_text(conftest)
    checker = git_show(repo, ref, "scripts/ci/junit_timeouts.py")
    if checker is not None:
        (root / "scripts/ci").mkdir(parents=True)
        (root / "scripts/ci/junit_timeouts.py").write_text(checker)


def stale_green_junit(root: Path, out: Path) -> None:
    """JUnit files from an all-pass run, as an earlier job on a reused runner would leave."""
    for name in ("gw01-a.xml", "gw01-b.xml"):
        subprocess.run(
            [VENV_PY, "-m", "pytest", "-q", "-p", "no:cacheprovider", "-k", "test_ok",
             "tests/openai_conformance", f"--junitxml={out / name}"],
            cwd=root / "gateway_v2", capture_output=True, text=True, check=False,
        )


KEYS = ("junit_timeouts", "lgw01", "drift", "crashed", "LGW01", "Traceback",
        "FileNotFoundError", "passed", "failed", "Timeout (>")


def run(repo: str, ref: str, scenario: str) -> tuple[str, int, float, list[str]]:
    script = step_text(repo, ref)
    with tempfile.TemporaryDirectory(prefix="lgw01-") as tmp:
        root = Path(tmp) / "repo"
        out = Path(tmp) / "out"
        out.mkdir()
        build(root, repo, ref, scenario)
        script = script.replace("/tmp/gw01-", f"{out}/gw01-")
        if SCENARIOS[scenario][2]:
            stale_green_junit(root, out)
        (Path(tmp) / "step.sh").write_text(script)
        env = {
            "PATH": os.environ["PATH"],
            "HOME": os.environ.get("HOME", "/root"),
            "PYTHONPATH": str(root / "gateway_v2"),
            "HARNESS_FLAG": str(Path(tmp) / "hang-once.flag"),
            "PYTHONDONTWRITEBYTECODE": "1",
        }
        start = time.monotonic()
        proc = subprocess.run(
            ["bash", "--noprofile", "--norc", "-eo", "pipefail", str(Path(tmp) / "step.sh")],
            cwd=root / "gateway_v2", env=env, capture_output=True, text=True, timeout=300,
        )
        took = time.monotonic() - start
        lines = [
            ln.rstrip() for ln in (proc.stdout + proc.stderr).splitlines()
            if any(k in ln for k in KEYS) and "PytestDeprecationWarning" not in ln
        ]
        lines = [ln.replace(str(out), "<tmp>").replace(str(root), "<root>") for ln in lines]
        return ("GREEN" if proc.returncode == 0 else "RED"), proc.returncode, took, lines


def main() -> int:
    repo, refs = sys.argv[1], sys.argv[2:]
    only = os.environ.get("HARNESS_ONLY")
    names = [n for n in SCENARIOS if not only or n in only.split(",")]
    table: dict[str, dict[str, str]] = {n: {} for n in names}
    for ref in refs:
        for name in names:
            verdict, code, took, lines = run(repo, ref, name)
            table[name][ref] = f"{verdict} ({code})"
            print(f"=== {ref[:8]} {name}: step {verdict}, exit {code}, {took:.1f} s")
            for ln in lines[-8:]:
                print(f"    {ln}")
            sys.stdout.flush()
    print()
    print("| scenario | " + " | ".join(r[:8] for r in refs) + " |")
    print("|---|" + "---|" * len(refs))
    for name in names:
        print(f"| {name} | " + " | ".join(table[name][r] for r in refs) + " |")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

def run_tests(root: Path) -> tuple[int, str]:
    proc = subprocess.run(
        [sys.executable, "-m", "unittest", "discover", "-s", str(root / "scripts/ci"),
         "-p", "test_junit_timeouts.py"],
        capture_output=True, text=True,
    )
    tail = [ln for ln in proc.stderr.splitlines() if ln.startswith(("Ran ", "OK", "FAILED"))]
    return proc.returncode, " ".join(tail)


def main() -> int:
    wt = Path(sys.argv[1])
    bad = 0
    with tempfile.TemporaryDirectory() as tmp:
        root = Path(tmp)
        for rel in (CHECKER, "scripts/ci/test_junit_timeouts.py", WF):
            (root / rel).parent.mkdir(parents=True, exist_ok=True)
            shutil.copy(wt / rel, root / rel)
        code, tail = run_tests(root)
        print(f"unmutated: exit {code}: {tail}")
        bad += code != 0
        for name, rel, old, new in MUTATIONS:
            original = (wt / rel).read_text()
            assert original.count(old) == 1, (name, original.count(old))
            (root / rel).write_text(original.replace(old, new))
            code, tail = run_tests(root)
            print(f"{'killed' if code else 'SURVIVED'}: {name}: {tail}")
            bad += code == 0
            (root / rel).write_text(original)
    print(f"mutations: {len(MUTATIONS)}, survived: {bad}")
    return 1 if bad else 0


if __name__ == "__main__":
    raise SystemExit(main())
